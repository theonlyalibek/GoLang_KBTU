package main

import (
    "fmt"
    "strings"
)

func WordCount(s string) map[string]int {
    counts := make(map[string]int)

    words := strings.Fields(s)

    for _, word := range words {
        counts[word] = counts[word] + 1
    }

    return counts
}

func main() {
    text := "раз два три три два раз"
    fmt.Println(WordCount(text))
}
