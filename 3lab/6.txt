package main

import (
	"fmt"
	"math"
)

type Shape interface {
	Area() float64
	Perimeter() float64
}

type Rectangle struct {
	Width, Height float64
}

func (r Rectangle) Area() float64 {
	return r.Width * r.Height
}

func (r Rectangle) Perimeter() float64 {
	return 2 * (r.Width + r.Height)
}

func (r *Rectangle) Scale(k float64) {
	r.Width *= k
	r.Height *= k
}

type Circle struct {
	Radius float64
}

func (c *Circle) Area() float64 {
	return math.Pi * c.Radius * c.Radius
}

func (c *Circle) Perimeter() float64 {
	return 2 * math.Pi * c.Radius
}

func main() {
	r := Rectangle{Width: 3, Height: 4}

	fmt.Println("Before Scale:", r)
	r.Scale(2)
	fmt.Println("After Scale:", r)

	c := &Circle{Radius: 1}
	shapes := []Shape{r, c}

	for _, shape := range shapes {
		fmt.Printf("%T: area=%.2f perimeter=%.2f\n",
			shape, shape.Area(), shape.Perimeter())
	}

	// декоммент для демонстрации ошибки
	// Circle не реализует Shape, потому что методы
	// Area и Perimeter объявлены для *Circle.
	// shapes = append(shapes, Circle{Radius: 1})
}
