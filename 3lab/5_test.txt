package main

import "testing"

var benchmarkResult float64

func BenchmarkFirstValue(b *testing.B) {
	r := Report{Title: "Example"}
	r.Values[0] = 42

	b.ReportAllocs()
	b.ResetTimer()

	for i := 0; i < b.N; i++ {
		benchmarkResult = r.FirstValue()
	}
}
